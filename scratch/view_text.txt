SELECT TEMP2.BASE, TEMP2.CODEMP,TEMP2.CODCENCUS,TEMP2.CODNAT,TEMP2.DTCOMP,TEMP2.DTREF,TEMP2.NUNOTA,TEMP2.CODPARC,TEMP2.REF, TEMP2.VALOR AS VALOR, TEMP2.ORCAMENTO FROM (
SELECT * FROM (
SELECT
     'COM' AS BASE
     ,CAB.CODEMP
    ,COALESCE(CAB.CODCENCUS,0)AS CODCENCUS
    ,CAB.CODNAT AS codnat
    ,CAB.DTENTSAI  AS DTCOMP
    ,TRUNC(DTENTSAI,'MM')  AS DTREF
    ,CAB.NUNOTA
    ,CAB.CODPARC
    ,SUM(((ITE.VLRTOT - ITE.VLRDESC - ITE.VLRREPRED + ITE.VLRSUBST + ITE.VLRIPI) * VCA.INDITENSBRUTO) ) AS valornf
    --Nessa linha o 1º case foi criado para buscar o custo da operação da china, que fica direto na tabela de item.
    ,SUM(CASE WHEN CAB.CODTIPOPER  IN (4092,3320,4091) THEN (ITE.AD_CUSTO_OPER_CHINA* ITE.QTDNEG)*CAB.VLRMOEDA ELSE
        	COALESCE(CASE WHEN (CAB.CODEMP <> 501 OR TOP.CODTIPOPER=3291) THEN (F_OBTEM_CUSTO_MQ ((CASE WHEN TOP.CODTIPOPER=3291 THEN 2 ELSE CAB.CODEMP END), ITE.CODPROD, CAB.DTNEG, CAB.DTNEG) * ITE.QTDNEG) END * (CASE WHEN CAB.CODTIPOPER=11000 THEN 0 ELSE 1 END) ,0)
 		END      
        ) AS CUSTOGER
    --Aqui a função foi trocada para pegar o custo sem impostos (F_OBTEM_CUSTO_MQ (CAB.CODEMP, ITE.CODPROD, CAB.DTNEG, CAB.DTNEG) ) não configurado Custo sem ICMS para FCC
    ,SUM(CASE WHEN CAB.CODEMP <> 501 THEN (OBTEMCUSTO(ITE.CODPROD,'S',CAB.CODEMP,'N',0,'N',0,CAB.DTENTSAI,3) * ITE.QTDNEG) END 
    	    * (CASE WHEN CAB.CODTIPOPER=11000 THEN 0 ELSE 1 END)
       ) AS CUSSIMICM
    ,SUM(F_OBTEM_IMPOSTOSITE(CAB.NUNOTA,ITE.SEQUENCIA)) AS IMPOSTO
    ,0 AS ORCAMENTO
        FROM TGFCAB CAB,
             TGFITE ITE,
             TGFTOP TOP,
             VGFCAB VCA
       WHERE     CAB.CODTIPOPER = TOP.CODTIPOPER
              AND CAB.DHTIPOPER = TOP.DHALTER
              AND CAB.NUNOTA = ITE.NUNOTA
              AND CAB.TIPMOV IN ('V', 'D')
              AND CAB.STATUSNOTA = 'L' 
              AND CAB.NUNOTA = VCA.NUNOTA
              AND (TOP.GOLSINAL = -1 OR BONIFICACAO='S' OR CAB.CODTIPOPER IN (3218,3327))
              AND CAB.STATUSNOTA = 'L'
              AND (NOT EXISTS(SELECT 1 FROM TGFVAR VAR WHERE VAR.NUNOTA = ITE.NUNOTA AND VAR.NUNOTAORIG = VAR.NUNOTA AND VAR.SEQUENCIAORIG = ITE.SEQUENCIA))
              AND (TOP.ATUALFINTERC <> 'N' OR TOP.ATUALESTTERC = 'N' OR  ITE.TERCEIROS <> 'S')
              AND ITE.SEQUENCIA > 0    
              AND TOP.GOLDEV IN (1,-1) 
              AND CAB.DTENTSAI>=TO_DATE('01/01/2024','DD/MM/YYYY')
              --    AND CAB.DTENTSAI<='30/04/2024'
              --    AND CAB.CODCENCUS NOT IN (21010000)
              --    AND CAB.CODEMP IN (7,8)
GROUP BY CAB.CODEMP, CAB.CODCENCUS, CAB.CODNAT, CAB.DTENTSAI, CAB.NUNOTA, CAB.CODPARC
) temp
UNPIVOT (valor FOR REF IN (valornf, custoger,  CUSSIMICM, IMPOSTO))
)TEMP2 
--INNER JOIN AD_CADDREDET DRE ON (DRE.REF=TEMP2.REF AND DRE.CODCENCUS=TEMP2.CODCENCUS AND DRE.CODNAT=TEMP2.CODNAT)

UNION ALL

/*    03/02/2026: Solicitado pela Iara, para incluir a natureza: 12020408-Bonus Premiacação Trimestral, para voltar ao mês anterior
    			similar a comissão, mas existe lançamentos feitos antes de 2026, se for atualizado pode dar dif

SELECT TEMPFIN.BASE, TEMPFIN.CODEMP,TEMPFIN.CODCENCUS,TEMPFIN.CODNAT,TEMPFIN.DTCOMP,TEMPFIN.DTREF,TEMPFIN.NUNOTA,TEMPFIN.CODPARC,TEMPFIN.REF, TEMPFIN.VALOR AS VALOR, TEMPFIN.ORCAMENTO  FROM (
SELECT * FROM (
SELECT 
        'FIN' AS BASE
        ,FIN.CODEMP
        ,fin.codcencus AS codcencus
        ,fin.codnat AS codnat
        --Criada condição especial para Comissões considerar o mês anterior do lançamento
        ,CASE WHEN fin.codnat IN (2010101,2010116,2010120,12010201,12010401, 12020408) THEN trunc(fin.dtneg,'MM')-1  ELSE fin.DTNEG END  AS DTCOMP
        ,TRUNC((CASE WHEN fin.codnat IN (2010101,2010116,2010120,12010201,12010401,12020408) THEN trunc(fin.dtneg,'MM')-1  ELSE fin.DTNEG END),'MM')  AS DTREF
        ,FIN.NUFIN AS NUNOTA
        ,(fin.VLRDESDOB*fin.recdesp) AS valorfin
        ,FIN.CODPARC
        ,0 AS ORCAMENTO
FROM VGFFINRAT FIN 
WHERE FIN.recdesp IN (1,-1)
AND FIN.provisao='N'
AND fin.dtneg>=TO_DATE('01/01/2024','DD/MM/YYYY')
--AND FIN.DHBAIXA<='30/04/2024'
--AND fin.codemp IN (7,8)
) TEMPFIN
UNPIVOT (valor FOR REF IN (VALORFIN))
WHERE valor<>0
)TEMPFIN 
--INNER JOIN AD_CADDREDET DRE ON (DRE.REF=TEMPFIN.REF AND DRE.CODCENCUS=TEMPFIN.CODCENCUS AND DRE.CODNAT=TEMPFIN.CODNAT)

UNION ALL

--Base FIN ACRESC/DESC
SELECT TEMP2.BASE, TEMP2.CODEMP,TEMP2.CODCENCUS,TEMP2.CODNAT,TEMP2.DTCOMP,TEMP2.DTREF,TEMP2.NUNOTA,TEMP2.CODPARC,TEMP2.REF, TEMP2.VALOR AS VALOR, TEMP2.ORCAMENTO  FROM (
SELECT * FROM (
SELECT 
        'FINACRESDESC' AS BASE
        ,FIN.CODEMP
        ,fin.codcencus AS codcencus
        ,fin.codnat AS codnat
        --Criada condição especial para Comissões considerar o mês anterior do lançamento
        ,CASE WHEN fin.codnat IN (2010101,2010116,2010120,12010201,12010401,12020408) THEN trunc(fin.DHBAIXA,'MM')-1  ELSE fin.DHBAIXA END  AS DTCOMP
        ,TRUNC((CASE WHEN fin.codnat IN (2010101,2010116,2010120,12010201,12010401,12020408) THEN trunc(fin.DHBAIXA,'MM')-1  ELSE fin.DHBAIXA END),'MM') AS DTREF
        ,FIN.NUFIN AS NUNOTA
        ,((fin.vlrdesc)*fin.recdesp*(-1)) * CASE WHEN (fin.VLRDESDOB>0 AND FIN.VLRATUALIZADO>0) THEN (fin.VLRDESDOB/FIN.VLRATUALIZADO) ELSE 1 END AS CARTAODESC
        ,((fin.vlrmulta+fin.vlrjuro)*fin.recdesp)  * CASE WHEN (fin.VLRDESDOB>0 AND FIN.VLRATUALIZADO>0) THEN (fin.VLRDESDOB/FIN.VLRATUALIZADO) ELSE 1 END AS FINACRESC
        ,((fin.cartaodesc)*fin.recdesp*(-1)) * CASE WHEN (fin.VLRDESDOB>0 AND FIN.VLRATUALIZADO>0) THEN (fin.VLRDESDOB/FIN.VLRATUALIZADO) ELSE 1 END AS TAXAADM
        ,(fin.vlrvarcambial*fin.recdesp) AS VARCAMBIAL
        ,FIN.CODPARC
        ,0 AS ORCAMENTO
FROM VGFFINRAT FIN 
WHERE FIN.recdesp IN (1,-1)
AND FIN.provisao='N'
AND FIN.DHBAIXA>=TO_DATE('01/01/2024','DD/MM/YYYY')
AND fin.codcencus NOT IN (20010000, 21010000)
--AND FIN.DHBAIXA<='30/04/2024'
--AND fin.codemp IN (7,8)
) temp
UNPIVOT (valor FOR REF IN (CARTAODESC, FINACRESC, TAXAADM, VARCAMBIAL))
WHERE valor<>0
)TEMP2 
--INNER JOIN AD_CADDREDET DRE ON (DRE.REF=TEMP2.REF AND DRE.CODCENCUS=TEMP2.CODCENCUS AND DRE.CODNAT=TEMP2.CODNAT)

UNION ALL

--IncQlik
SELECT TEMP2.BASE, TEMP2.CODEMP,TEMP2.CODCENCUS,TEMP2.CODNAT,TEMP2.DTCOMP,TEMP2.DTREF,TEMP2.NUNOTA,TEMP2.CODPARC,TEMP2.REF, TEMP2.VALOR AS VALOR, TEMP2.ORCAMENTO FROM (
SELECT * FROM (
SELECT  
    'INC' AS BASE
    ,CODEMP
    ,codcencus AS codcencus
    ,codnat AS codnat
    ,DTNEG AS DTCOMP
    ,TRUNC(DTNEG,'MM')  AS DTREF
    ,IDDRE AS NUNOTA
    ,CASE WHEN tipoitem<>'ORC' THEN VALOR ELSE 0 END AS INCQLIK
    ,0 AS CODPARC
    ,0 AS ORCAMENTO
FROM AD_FLUXODRE
WHERE DTNEG>=TO_DATE('01/01/2024','DD/MM/YYYY')
--AND DTNEG<='30/04/2024'
--AND CODEMP IN (7,8)
) temp
UNPIVOT (valor FOR REF IN (INCQLIK))
)TEMP2 
--INNER JOIN AD_CADDREDET DRE ON (DRE.REF=TEMP2.REF AND DRE.CODCENCUS=TEMP2.CODCENCUS AND DRE.CODNAT=TEMP2.CODNAT)
--and DTREF>='01/09/2025'

UNION ALL 

SELECT  
    CASE 
    	WHEN REF IN ('VALORFIN', 'CARTAODESC', 'TAXAADM','FINACRESC') THEN 'FIN' 
        WHEN REF='INCQLIK' THEN 'INC'  
        ELSE 'COM' END AS BASE --TEMP2.BASE
    ,501 AS CODEMP --, TEMP2.CODEMP
    ,codcencus AS codcencus --,TEMP2.CODCENCUS
    ,codnat AS codnat --,TEMP2.CODNAT
    ,DTREF AS DTCOMP --,TEMP2.DTCOMP
    ,DTREF AS DTREF --,TEMP2.DTREF
    ,IDORC AS NUNOTA --,TEMP2.NUNOTA
    ,0 AS CODPARC --,TEMP2.CODPARC
    ,REF --,TEMP2.REF
    ,0 AS VALOR --, TEMP2.VALOR * DRE.FATOR AS VALOR
    ,VLRORC AS ORCAMENTO   --, TEMP2.ORCAMENTO 

FROM AD_ORCAMENTO
WHERE CODCENCUS NOT IN (21010000)
